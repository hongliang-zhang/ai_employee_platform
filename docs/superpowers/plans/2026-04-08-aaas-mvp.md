# Agent as a Service MVP Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the AaaS MVP — a platform where users deploy AI agents accessible via Telegram, with conversation history persisted across e2b sandbox restarts.

**Architecture:** A pnpm monorepo with three packages (`backend`, `agent-sdk`, `dashboard`) and one agent template (`hello-agent`). `backend` is a single Node.js process exposing two HTTP servers: port 3001 (internal-api for dashboard) and port 3002 (gateway-api for sandboxes). e2b sandboxes only receive a JWT + gateway URL — no credentials.

**Tech Stack:** Node.js 20, TypeScript 5, Express 4, `pg` (node-postgres), `jsonwebtoken`, `@e2b/code-interpreter`, Next.js 14, Vitest, `msw` (HTTP mocking)

`packages/backend/src/db.ts`:
```typescript
import pg from 'pg'
import fs from 'node:fs'
import path from 'node:path'
import { fileURLToPath } from 'node:url'

const __dirname = path.dirname(fileURLToPath(import.meta.url))

let pool: pg.Pool | null = null

export function getPool(): pg.Pool {
  if (!pool) {
    pool = new pg.Pool({ connectionString: process.env.DATABASE_URL })
  }
  return pool
}

export async function runMigrations(): Promise<void> {
  const client = await getPool().connect()
  try {
    await client.query(`
      CREATE TABLE IF NOT EXISTS _migrations (
        filename TEXT PRIMARY KEY,
        applied_at TIMESTAMPTZ NOT NULL DEFAULT now()
      )
    `)

    const migrationsDir = path.join(__dirname, '..', 'migrations')
    const files = fs.readdirSync(migrationsDir).filter(f => f.endsWith('.sql')).sort()

    for (const file of files) {
      const { rows } = await client.query(
        'SELECT 1 FROM _migrations WHERE filename = $1', [file]
      )
      if (rows.length > 0) continue

      const sql = fs.readFileSync(path.join(migrationsDir, file), 'utf8')
      await client.query('BEGIN')
      await client.query(sql)
      await client.query('INSERT INTO _migrations (filename) VALUES ($1)', [file])
      await client.query('COMMIT')
      console.log(`[migrations] applied: ${file}`)
    }
  } finally {
    client.release()
  }
}
```

```typescript
import jwt from 'jsonwebtoken'

export interface JwtPayload {
  conversation_id: string
  agent_id: string
  jti: string
  exp?: number
}

export function signJwt(
  payload: Omit<JwtPayload, 'exp'>,
  secret: string,
  expiresInSeconds: number
): string {
  return jwt.sign(payload, secret, {
    algorithm: 'HS256',
    expiresIn: expiresInSeconds,
  })
}

export function verifyJwt(token: string, secret: string): JwtPayload {
  return jwt.verify(token, secret) as JwtPayload
}
```

```typescript
import crypto from 'node:crypto'

const ALGORITHM = 'aes-256-gcm'
const IV_LEN = 12
const AUTH_TAG_LEN = 16

function keyBuffer(key: string): Buffer {
  // Key must be exactly 32 bytes for AES-256
  return Buffer.from(key.slice(0, 32).padEnd(32, '0'))
}

export function encryptToken(plaintext: string, key: string): string {
  const iv = crypto.randomBytes(IV_LEN)
  const cipher = crypto.createCipheriv(ALGORITHM, keyBuffer(key), iv)
  const encrypted = Buffer.concat([cipher.update(plaintext, 'utf8'), cipher.final()])
  const authTag = cipher.getAuthTag()
  // Format: iv(hex):authTag(hex):ciphertext(hex)
  return [iv.toString('hex'), authTag.toString('hex'), encrypted.toString('hex')].join(':')
}

export function decryptToken(ciphertext: string, key: string): string {
  const [ivHex, authTagHex, encHex] = ciphertext.split(':')
  const iv = Buffer.from(ivHex, 'hex')
  const authTag = Buffer.from(authTagHex, 'hex')
  const encrypted = Buffer.from(encHex, 'hex')
  const decipher = crypto.createDecipheriv(ALGORITHM, keyBuffer(key), iv)
  decipher.setAuthTag(authTag)
  return decipher.update(encrypted).toString('utf8') + decipher.final('utf8')
}
```

```typescript
import { getPool } from '../db.js'

export interface ContentPart {
  type: 'text'
  text: string
}

export interface StoredMessage {
  id?: string
  role: 'user' | 'assistant' | 'system' | 'tool'
  content: ContentPart[]
  source: 'im' | 'web' | 'sandbox' | 'system'
  external_message_id?: string | null
  metadata?: Record<string, unknown>
  created_at?: string
}

export interface AppendedMessage {
  id: string
  created_at: string
}

export async function loadMessages(
  conversationId: string,
  afterMessageId?: string | null
): Promise<{ messages: StoredMessage[]; last_message_id: string | null }> {
  const pool = getPool()
  let query: string
  let params: unknown[]

  if (afterMessageId) {
    const { rows: [anchor] } = await pool.query(
      'SELECT created_at FROM messages WHERE id = $1', [afterMessageId]
    )
    query = `SELECT * FROM messages WHERE conversation_id = $1 AND created_at > $2 ORDER BY created_at ASC`
    params = [conversationId, anchor?.created_at ?? new Date(0)]
  } else {
    query = `SELECT * FROM messages WHERE conversation_id = $1 ORDER BY created_at ASC`
    params = [conversationId]
  }

  const { rows } = await pool.query(query, params)
  const messages = rows.map((r: any) => ({
    id: r.id,
    role: r.role,
    content: r.content_json,
    source: r.source,
    external_message_id: r.external_message_id,
    metadata: r.metadata_json,
    created_at: r.created_at,
  }))
  const last_message_id = messages.length > 0 ? messages[messages.length - 1].id! : null
  return { messages, last_message_id }
}

export async function appendMessages(
  conversationId: string,
  expectedLastMessageId: string | null,
  messages: Omit<StoredMessage, 'id' | 'created_at'>[]
): Promise<{ appended: AppendedMessage[]; last_message_id: string }> {
  const pool = getPool()
  const client = await pool.connect()
  try {
    await client.query('BEGIN')

    // Optimistic concurrency check
    const { rows } = await client.query(
      `SELECT id FROM messages WHERE conversation_id = $1 ORDER BY created_at DESC LIMIT 1`,
      [conversationId]
    )
    const actualLastId = rows[0]?.id ?? null

    if (expectedLastMessageId !== actualLastId) {
      throw { code: 'stale_write', actual_last_message_id: actualLastId }
    }

    const appended: AppendedMessage[] = []
    for (const msg of messages) {
      const { rows: [inserted] } = await client.query(
        `INSERT INTO messages
           (conversation_id, role, content_json, source, external_message_id, metadata_json)
         VALUES ($1, $2, $3, $4, $5, $6)
         RETURNING id, created_at`,
        [
          conversationId, msg.role, JSON.stringify(msg.content),
          msg.source, msg.external_message_id ?? null,
          JSON.stringify(msg.metadata ?? {})
        ]
      )
      appended.push({ id: inserted.id, created_at: inserted.created_at })
    }

    await client.query(
      `UPDATE conversations SET last_message_at = now() WHERE id = $1`, [conversationId]
    )

    await client.query('COMMIT')
    return { appended, last_message_id: appended[appended.length - 1].id }
  } catch (err) {
    await client.query('ROLLBACK')
    throw err
  } finally {
    client.release()
  }
}
```

```typescript
export interface LlmRequest {
  model: string
  messages: Array<{ role: string; content: Array<{ type: string; text?: string }> }>
  tools?: unknown[]
  tool_choice?: string | object
}

export interface LlmResponse {
  provider: string
  model: string
  message: {
    role: string
    content: Array<{ type: 'text'; text: string }>
    tool_calls: unknown[]
  }
  finish_reason: string
  usage: { input_tokens: number; output_tokens: number; total_tokens: number }
  request_id: string
}

export async function proxyLlm(req: LlmRequest, apiKey: string): Promise<LlmResponse> {
  const response = await fetch('https://api.openai.com/v1/chat/completions', {
    method: 'POST',
    headers: {
      'Authorization': `Bearer ${apiKey}`,
      'Content-Type': 'application/json',
    },
    body: JSON.stringify({
      model: req.model,
      messages: req.messages.map(m => ({
        role: m.role,
        content: m.content.map(c => c.text).join(''),
      })),
      tools: req.tools,
      tool_choice: req.tool_choice,
    }),
  })

  if (!response.ok) {
    throw { code: 'provider_error', status: response.status }
  }

  const data = await response.json() as any
  const choice = data.choices[0]

  return {
    provider: 'openai',
    model: data.model,
    message: {
      role: choice.message.role,
      content: choice.message.content
        ? [{ type: 'text' as const, text: choice.message.content }]
        : [],
      tool_calls: choice.message.tool_calls ?? [],
    },
    finish_reason: choice.finish_reason,
    usage: {
      input_tokens: data.usage.prompt_tokens,
      output_tokens: data.usage.completion_tokens,
      total_tokens: data.usage.total_tokens,
    },
    request_id: data.id ?? '',
  }
}
```

```typescript
import { Router, Request, Response, NextFunction } from 'express'
import { verifyJwt } from './jwt.js'
import { loadMessages, appendMessages } from './messages.js'
import { proxyLlm } from './llm.js'

export function createGatewayRouter(): Router {
  const router = Router()
  router.use(express.json())

  // JWT auth middleware
  router.use((req: Request, res: Response, next: NextFunction) => {
    const auth = req.headers.authorization
    if (!auth?.startsWith('Bearer ')) {
      return res.status(401).json({ error: { code: 'unauthorized', message: 'Missing token', retryable: false, details: {} } })
    }
    const token = auth.slice(7)
    try {
      const secret = process.env.JWT_SECRET!
      const payload = verifyJwt(token, secret)
      ;(req as any).jwtPayload = payload
      next()
    } catch (err: any) {
      const code = err.message === 'jwt expired' ? 'token_expired' : 'unauthorized'
      return res.status(401).json({ error: { code, message: err.message, retryable: false, details: {} } })
    }
  })

  router.post('/gateway/llm', async (req: Request, res: Response) => {
    try {
      const result = await proxyLlm(req.body, process.env.OPENAI_API_KEY!)
      res.json(result)
    } catch (err: any) {
      res.status(502).json({ error: { code: err.code ?? 'provider_error', message: String(err), retryable: true, details: {} } })
    }
  })

  router.post('/gateway/messages/load', async (req: Request, res: Response) => {
    const { conversation_id } = (req as any).jwtPayload
    try {
      const result = await loadMessages(conversation_id, req.body?.after_message_id)
      res.json({ conversation_id, ...result })
    } catch (err) {
      res.status(500).json({ error: { code: 'internal_error', message: String(err), retryable: false, details: {} } })
    }
  })

  router.post('/gateway/messages/append', async (req: Request, res: Response) => {
    const { conversation_id } = (req as any).jwtPayload
    const { expected_last_message_id, messages } = req.body
    try {
      const result = await appendMessages(conversation_id, expected_last_message_id, messages)
      res.json({ conversation_id, ...result })
    } catch (err: any) {
      if (err.code === 'stale_write') {
        return res.status(409).json({ error: { code: 'stale_write', message: 'expected_last_message_id mismatch', retryable: false, details: { actual_last_message_id: err.actual_last_message_id } } })
      }
      res.status(500).json({ error: { code: 'internal_error', message: String(err), retryable: false, details: {} } })
    }
  })

  router.post('/gateway/files/presign', (_req: Request, res: Response) => {
    res.status(501).json({ error: { code: 'not_implemented', message: 'File storage not yet available', retryable: false, details: {} } })
  })

  return router
}
```

```typescript
import type { LlmRequest, LlmResponse, LoadMessagesResult, AppendMessagesResult, StoredMessage } from './types.js'

export interface GatewayOptions {
  token?: string
  gatewayUrl?: string
  sessionId?: string
}

export class Gateway {
  private token: string
  private gatewayUrl: string
  private sessionId: string

  constructor(opts: GatewayOptions = {}) {
    this.token = opts.token ?? process.env.SESSION_TOKEN ?? ''
    this.gatewayUrl = (opts.gatewayUrl ?? process.env.GATEWAY_URL ?? '').replace(/\/$/, '')
    this.sessionId = opts.sessionId ?? process.env.SESSION_ID ?? ''
  }

  private async post<T>(path: string, body: unknown): Promise<T> {
    const res = await fetch(`${this.gatewayUrl}${path}`, {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${this.token}`,
        'Content-Type': 'application/json',
      },
      body: JSON.stringify(body),
    })
    const data = await res.json() as any
    if (!res.ok) throw data.error ?? { code: 'unknown_error', status: res.status }
    return data as T
  }

  async loadMessages(afterMessageId?: string | null): Promise<LoadMessagesResult> {
    return this.post('/gateway/messages/load', { after_message_id: afterMessageId ?? undefined })
  }

  async appendMessages(params: {
    expected_last_message_id: string | null
    messages: Omit<StoredMessage, 'id' | 'created_at'>[]
  }): Promise<AppendMessagesResult> {
    return this.post('/gateway/messages/append', params)
  }

  async invokeLlm(req: LlmRequest): Promise<LlmResponse> {
    return this.post('/gateway/llm', req)
  }
}
```

```typescript
import { Sandbox } from '@e2b/code-interpreter'
import { sandboxMap } from './sandbox-map.js'
import { signJwt } from '../gateway/jwt.js'
import { getPool } from '../db.js'

// Maps conversation_id -> agent_id for killAllForAgent lookups
const conversationAgentMap = new Map<string, string>()

export async function dispatch(params: {
  conversationId: string
  agentId: string
  templateId: string
  port: number
  idleTimeoutMs: number
  message: string
}): Promise<{ reply: string }> {
  const { conversationId, agentId, templateId, port, idleTimeoutMs, message } = params
  const secret = process.env.JWT_SECRET!
  const gatewayUrl = process.env.GATEWAY_URL!

  conversationAgentMap.set(conversationId, agentId)

  // Sign a fresh JWT on every dispatch (handles long-running conversations)
  const newToken = signJwt(
    { conversation_id: conversationId, agent_id: agentId, jti: `sess-${Date.now()}` },
    secret,
    Math.ceil(idleTimeoutMs / 1000) + 300
  )

  let entry = sandboxMap.get(conversationId)

  if (entry) {
    // Refresh token on existing sandbox before calling /chat
    const refreshRes = await fetchSandbox(entry.sandboxId, port, '/refresh-token', { token: newToken })
    if (!refreshRes.ok) {
      // Sandbox is unhealthy — kill it and fall through to create a new one
      await killSandbox(conversationId)
      entry = undefined
    } else {
      entry.sessionToken = newToken
    }
  }

  if (!entry) {
    // Cold start: create new sandbox
    const sandbox = await Sandbox.create(templateId, {
      envs: {
        SESSION_TOKEN: newToken,
        GATEWAY_URL: gatewayUrl,
        SESSION_ID: conversationId,
        PORT: String(port),
      },
    })

    // Poll /health until ready (max 30s)
    await waitForHealth(sandbox.sandboxId, port, 30_000)

    const idleTimer = setTimeout(() => killSandbox(conversationId), idleTimeoutMs)
    entry = { sandboxId: sandbox.sandboxId, sessionToken: newToken, idleTimer }
    sandboxMap.set(conversationId, entry)
  }

  // Reset idle timer
  clearTimeout(entry.idleTimer)
  entry.idleTimer = setTimeout(() => killSandbox(conversationId), idleTimeoutMs)

  // Call /chat
  const chatRes = await fetchSandbox(entry.sandboxId, port, '/chat', { message })
  if (!chatRes.ok) {
    throw new Error(`Sandbox /chat returned ${chatRes.status}`)
  }
  const body = await chatRes.json() as { reply: string }
  return { reply: body.reply }
}

async function waitForHealth(sandboxId: string, port: number, timeoutMs: number): Promise<void> {
  const deadline = Date.now() + timeoutMs
  while (Date.now() < deadline) {
    try {
      const res = await fetchSandbox(sandboxId, port, '/health', null, 'GET')
      if (res.ok) return
    } catch {}
    await new Promise(r => setTimeout(r, 500))
  }
  throw new Error(`Sandbox ${sandboxId} did not become healthy within ${timeoutMs}ms`)
}

async function fetchSandbox(
  sandboxId: string,
  port: number,
  path: string,
  body: unknown,
  method = 'POST'
): Promise<Response> {
  // e2b sandbox URL pattern
  const url = `https://${port}-${sandboxId}.e2b.dev${path}`
  return fetch(url, {
    method,
    headers: body ? { 'Content-Type': 'application/json' } : {},
    body: body ? JSON.stringify(body) : undefined,
  })
}

export async function killSandbox(conversationId: string): Promise<void> {
  const entry = sandboxMap.get(conversationId)
  if (!entry) return
  sandboxMap.delete(conversationId)
  try {
    await Sandbox.kill(entry.sandboxId)
  } catch {}
}

export async function killAllForAgent(agentId: string): Promise<void> {
  const convIds = [...sandboxMap.keys()].filter(
    cid => conversationAgentMap.get(cid) === agentId
  )
  await Promise.all(convIds.map(killSandbox))
}
```

```typescript
  channel_key: string
  external_chat_id: string
  external_thread_key: string
  external_message_id: string

export function normalizeTelegramUpdate(
  update: any,
  imConfigId: string
): NormalizedMessage | null {
  const msg = update.message
  if (!msg?.text) return null          // ignore non-text and non-message updates

    external_thread_key: msg.message_thread_id ? String(msg.message_thread_id) : '',
    external_message_id: String(msg.message_id),

      external_user_id: String(msg.from.id),
      display_name: msg.from.first_name ?? null,
    },
    content: { type: 'text', text: msg.text },
  }
```

```typescript
import { getPool } from '../db.js'
import { appendMessages } from '../gateway/messages.js'
import { dispatch } from '../sandbox-orchestrator/index.js'
import type { NormalizedMessage } from './normalize.js'

export async function routeMessage(
  msg: NormalizedMessage,
  agentId: string,
  agentTypeConfig: { templateId: string; port: number; idleTimeoutMs: number }
): Promise<void> {
  const pool = getPool()
  const instanceId = process.env.INSTANCE_ID ?? 'local'

  // 1. Upsert conversation (BEFORE inserting inbound_job — NOT NULL constraint)
  const { rows: [conv] } = await pool.query(
    `INSERT INTO conversations (agent_id, channel_key, external_chat_id, external_thread_key)
     VALUES ($1, $2, $3, $4)
     ON CONFLICT (channel_key, external_chat_id, external_thread_key)
     DO UPDATE SET last_message_at = now()
     RETURNING id`,
    [agentId, msg.channel_key, msg.external_chat_id, msg.external_thread_key]
  )
  const conversationId: string = conv.id

  // 2. Dedup: insert inbound_job — conflict means already processed
  const { rowCount } = await pool.query(
    `INSERT INTO inbound_jobs (channel_key, external_message_id, conversation_id)
     VALUES ($1, $2, $3)
     ON CONFLICT (channel_key, external_message_id) DO NOTHING`,
    [msg.channel_key, msg.external_message_id, conversationId]
  )
  if (rowCount === 0) return // duplicate — discard

  // 3. Claim processing lease
  await pool.query(
    `UPDATE inbound_jobs
     SET status = 'processing', lease_owner = $1, lease_expires_at = now() + interval '60 seconds'
     WHERE channel_key = $2 AND external_message_id = $3`,
    [instanceId, msg.channel_key, msg.external_message_id]
  )

  try {
    // 4. Append user message to canonical history
    const { last_message_id } = await appendMessages(conversationId, null, [{
      role: 'user',
      content: [{ type: 'text', text: msg.content.text }],
      source: 'im',
      external_message_id: msg.external_message_id,
    }])
    // Note: appendMessages with expected=null will fail if history is non-empty.
    // The correct expected_last_message_id should come from a prior loadMessages call.
    // For simplicity in MVP routing, we fetch current head first:
    // (see below for corrected implementation)
    void last_message_id // suppress unused warning — see corrected version below

    // 5. Dispatch to sandbox
    const { reply } = await dispatch({
      conversationId,
      agentId,
      ...agentTypeConfig,
      message: msg.content.text,
    })

    // 6. Mark done
    await pool.query(
      `UPDATE inbound_jobs SET status = 'done'
       WHERE channel_key = $1 AND external_message_id = $2`,
```

```typescript
import { normalizeTelegramUpdate } from '../conversation-router/normalize.js'
import type { NormalizedMessage } from '../conversation-router/normalize.js'

export interface TelegramPollerOptions {
  botToken: string
  imConfigId: string
  onMessage: (msg: NormalizedMessage) => Promise<void>
}

export class TelegramPoller {
  private running = false
  private offset = 0
  private opts: TelegramPollerOptions

  constructor(opts: TelegramPollerOptions) {
    this.opts = opts
  }

  start(): void {
    this.running = true
    this.poll()
  }

  stop(): void {
    this.running = false
  }

  private async poll(): Promise<void> {
    while (this.running) {
      try {
        const url = `https://api.telegram.org/bot${this.opts.botToken}/getUpdates?offset=${this.offset}&timeout=30`
        const res = await fetch(url)
        if (!res.ok) { await sleep(5000); continue }

        const data = await res.json() as { ok: boolean; result: any[] }
        if (!data.ok) { await sleep(5000); continue }

        for (const update of data.result) {
          this.offset = update.update_id + 1
          const normalized = normalizeTelegramUpdate(update, this.opts.imConfigId)
          if (normalized) {
            await this.opts.onMessage(normalized).catch(err =>
              console.error('[telegram-poller] onMessage error:', err)
            )
          }
        }
      } catch (err) {
        console.error('[telegram-poller] poll error:', err)
        await sleep(5000)
      }
    }
  }
}

function sleep(ms: number) { return new Promise(r => setTimeout(r, ms)) }
```

```typescript
import { TelegramPoller } from './telegram-poller.js'
import { routeMessage } from '../conversation-router/index.js'
import { decryptToken } from '../crypto.js'
import { getPool } from '../db.js'
import type { NormalizedMessage } from '../conversation-router/normalize.js'

interface BindingEntry {
  poller: TelegramPoller
  sendReply: (chatId: string, text: string) => Promise<void>
}

const activeBindings = new Map<string, BindingEntry>()

export async function startAll(): Promise<void> {
  const pool = getPool()
  const { rows } = await pool.query(
    `SELECT ic.id, ic.bot_token_enc, ic.agent_id,
            at.template_id, at.port, at.idle_timeout_ms
     FROM im_configs ic
     JOIN agents a ON a.id = ic.agent_id
     JOIN agent_types at ON at.id = a.agent_type_id
     WHERE ic.status = 'active' AND a.status = 'active'`
  )
  for (const row of rows) {
    await acquireBinding(row)
  }
}

export async function acquireBinding(row: {
  id: string; bot_token_enc: string; agent_id: string;
  template_id: string; port: number; idle_timeout_ms: number
}): Promise<void> {
  if (activeBindings.has(row.id)) return

  const botToken = decryptToken(row.bot_token_enc, process.env.BOT_TOKEN_ENCRYPTION_KEY!)
  const agentTypeConfig = { templateId: row.template_id, port: row.port, idleTimeoutMs: row.idle_timeout_ms }

  const sendReply = makeSendReply(botToken)

  const poller = new TelegramPoller({
    botToken,
    imConfigId: row.id,
    onMessage: async (msg: NormalizedMessage) => {
      try {
        await routeMessage(msg, row.agent_id, agentTypeConfig)
        const reply = (routeMessage as any)._lastReply
        if (reply) await sendReply(msg.external_chat_id, reply)
      } catch (err) {
        console.error('[im-ingress] routing error:', err)
        await sendReply(msg.external_chat_id, '抱歉，处理失败，请稍后重试').catch(() => {})
      }
    },
  })

  poller.start()
  activeBindings.set(row.id, { poller, sendReply })
}

export function releaseBinding(imConfigId: string): void {
  const entry = activeBindings.get(imConfigId)
  if (entry) {
    entry.poller.stop()
    activeBindings.delete(imConfigId)
  }
}

function makeSendReply(botToken: string) {
  return async (chatId: string, text: string): Promise<void> => {
    await fetch(`https://api.telegram.org/bot${botToken}/sendMessage`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ chat_id: chatId, text }),
    })
  }
}
```

```typescript
import { Router, Request, Response, NextFunction } from 'express'
import express from 'express'
import { createAgentsRouter } from './agents.js'
import { createImConfigsRouter } from './im-configs.js'
import { createChatRouter } from './chat.js'

export function createInternalRouter(): Router {
  const router = Router()
  router.use(express.json())

  // API key auth
  router.use((req: Request, res: Response, next: NextFunction) => {
    const key = req.headers['x-internal-api-key']
    if (key !== process.env.INTERNAL_API_KEY) {
      return res.status(401).json({ error: { code: 'unauthorized', message: 'Invalid API key' } })
    }
    next()
  })

  router.use(createAgentsRouter())
  router.use(createImConfigsRouter())
  router.use(createChatRouter())

  return router
}
```

```typescript
import { Router, Request, Response } from 'express'
import { getPool } from '../db.js'
import { acquireBinding, releaseBinding } from '../im-ingress/index.js'
import { killAllForAgent } from '../sandbox-orchestrator/index.js'

export function createAgentsRouter(): Router {
  const router = Router()

  router.post('/internal/agents', async (req: Request, res: Response) => {
    const { agent_type_id, name } = req.body
    if (!agent_type_id || !name) {
      return res.status(400).json({ error: { code: 'invalid_request', message: 'agent_type_id and name required' } })
    }
    const pool = getPool()
    const { rows: [agent] } = await pool.query(
      `INSERT INTO agents (agent_type_id, name) VALUES ($1, $2) RETURNING *`,
      [agent_type_id, name]
    )
    res.status(201).json(agent)
  })

  router.post('/internal/agents/:id/activate', async (req: Request, res: Response) => {
    const pool = getPool()
    await pool.query(`UPDATE agents SET status = 'active' WHERE id = $1`, [req.params.id])

    const { rows } = await pool.query(
      `SELECT ic.id, ic.bot_token_enc, ic.agent_id,
              at.template_id, at.port, at.idle_timeout_ms
       FROM im_configs ic
       JOIN agents a ON a.id = ic.agent_id
       JOIN agent_types at ON at.id = a.agent_type_id
       WHERE ic.agent_id = $1 AND ic.status = 'active'`,
      [req.params.id]
    )
    await Promise.all(rows.map(acquireBinding))
    res.json({ status: 'activated' })
  })

  router.post('/internal/agents/:id/deactivate', async (req: Request, res: Response) => {
    const pool = getPool()
    await pool.query(`UPDATE agents SET status = 'paused' WHERE id = $1`, [req.params.id])

    const { rows } = await pool.query(
      `SELECT id FROM im_configs WHERE agent_id = $1`, [req.params.id]
    )
    for (const { id } of rows) releaseBinding(id)
    await killAllForAgent(req.params.id)
    res.json({ status: 'deactivated' })
  })

  router.get('/internal/agents', async (_req: Request, res: Response) => {
    const { rows } = await getPool().query(
      `SELECT a.*, at.name as type_name FROM agents a JOIN agent_types at ON at.id = a.agent_type_id ORDER BY a.created_at DESC`
    )
    res.json(rows)
  })

  router.get('/internal/agent-types', async (_req: Request, res: Response) => {
    const { rows } = await getPool().query(`SELECT * FROM agent_types ORDER BY name`)
    res.json(rows)
  })

  return router
}
```

```typescript
import { Router, Request, Response } from 'express'
import { getPool } from '../db.js'
import { encryptToken, decryptToken } from '../crypto.js'
import { acquireBinding, releaseBinding } from '../im-ingress/index.js'

export function createImConfigsRouter(): Router {
  const router = Router()
  const encKey = () => process.env.BOT_TOKEN_ENCRYPTION_KEY!

  router.post('/internal/agents/:id/im-configs', async (req: Request, res: Response) => {
    const { platform = 'telegram', bot_token } = req.body
    if (!bot_token) {
      return res.status(400).json({ error: { code: 'invalid_request', message: 'bot_token required' } })
    }
    const pool = getPool()
    const bot_token_enc = encryptToken(bot_token, encKey())
    const { rows: [cfg] } = await pool.query(
      `INSERT INTO im_configs (agent_id, platform, bot_token_enc) VALUES ($1, $2, $3) RETURNING *`,
      [req.params.id, platform, bot_token_enc]
    )

    // If agent is active, start polling immediately
    const { rows: [agent] } = await pool.query(
      `SELECT a.status, at.template_id, at.port, at.idle_timeout_ms
       FROM agents a JOIN agent_types at ON at.id = a.agent_type_id WHERE a.id = $1`,
      [req.params.id]
    )
    if (agent?.status === 'active') {
      await acquireBinding({ ...cfg, agent_id: req.params.id, ...agent })
    }

    res.status(201).json({ im_config_id: cfg.id, status: 'active' })
  })

  router.put('/internal/im-configs/:id', async (req: Request, res: Response) => {
    const pool = getPool()
    const { bot_token } = req.body

    // Stop existing connection first
    releaseBinding(req.params.id)

    if (bot_token) {
      const bot_token_enc = encryptToken(bot_token, encKey())
      await pool.query(
        `UPDATE im_configs SET bot_token_enc = $1 WHERE id = $2`,
        [bot_token_enc, req.params.id]
      )
    }

    // Restart connection with updated config
    const { rows: [row] } = await pool.query(
      `SELECT ic.id, ic.bot_token_enc, ic.agent_id,
              at.template_id, at.port, at.idle_timeout_ms
       FROM im_configs ic
       JOIN agents a ON a.id = ic.agent_id
       JOIN agent_types at ON at.id = a.agent_type_id
       WHERE ic.id = $1`,
      [req.params.id]
    )
    if (row) await acquireBinding(row)
    res.json({ status: 'updated' })
  })

  router.delete('/internal/im-configs/:id', async (req: Request, res: Response) => {
    releaseBinding(req.params.id)
    await getPool().query(
      `UPDATE im_configs SET status = 'disabled' WHERE id = $1`, [req.params.id]
    )
    res.json({ status: 'deleted' })
  })

  return router
}
```

```typescript
import { Router, Request, Response } from 'express'
import { getPool } from '../db.js'
import { routeMessage } from '../conversation-router/index.js'
import type { NormalizedMessage } from '../conversation-router/normalize.js'

const SESSION_ID_RE = /^[a-zA-Z0-9_\-]{1,128}$/

export function createChatRouter(): Router {
  const router = Router()

  router.post('/internal/agents/:id/chat', async (req: Request, res: Response) => {
    const { session_id, message } = req.body

    if (!session_id || !SESSION_ID_RE.test(session_id)) {
      return res.status(400).json({ error: { code: 'invalid_request', message: 'session_id must be 1–128 chars [a-zA-Z0-9_-]' } })
    }
    if (!message || typeof message !== 'string') {
      return res.status(400).json({ error: { code: 'invalid_request', message: 'message required' } })
    }

    const pool = getPool()
    const { rows: [agent] } = await pool.query(
      `SELECT a.id, at.template_id, at.port, at.idle_timeout_ms
       FROM agents a JOIN agent_types at ON at.id = a.agent_type_id WHERE a.id = $1`,
      [req.params.id]
    )
    if (!agent) return res.status(404).json({ error: { code: 'not_found', message: 'Agent not found' } })

    const normalized: NormalizedMessage = {
      channel_key: `web:${req.params.id}:test`,
      external_chat_id: session_id,
      external_thread_key: '',
      external_message_id: `web-${Date.now()}`,
      author: { external_user_id: 'web-user', display_name: 'dashboard user' },
      content: { type: 'text', text: message },
    }

    try {
      await routeMessage(normalized, agent.id, {
        templateId: agent.template_id,
        port: agent.port,
        idleTimeoutMs: agent.idle_timeout_ms,
      })
      const reply = (routeMessage as any)._lastReply ?? ''
      const { rows: [conv] } = await pool.query(
        `SELECT id FROM conversations
         WHERE channel_key = $1 AND external_chat_id = $2 AND external_thread_key = ''`,
        [`web:${req.params.id}:test`, session_id]
      )
      res.json({ conversation_id: conv?.id, message: { role: 'assistant', content: [{ type: 'text', text: reply }] } })
    } catch (err) {
      res.status(500).json({ error: { code: 'internal_error', message: String(err) } })
    }
  })

  return router
}
```

```typescript
const BASE = process.env.BACKEND_URL ?? process.env.NEXT_PUBLIC_BACKEND_URL ?? 'http://localhost:3001'
const API_KEY = process.env.INTERNAL_API_KEY!

async function call<T>(method: string, path: string, body?: unknown): Promise<T> {
  const res = await fetch(`${BASE}${path}`, {
    method,
    headers: {
      'X-Internal-Api-Key': API_KEY,
      ...(body ? { 'Content-Type': 'application/json' } : {}),
    },
    body: body ? JSON.stringify(body) : undefined,
    cache: 'no-store',
  })
  if (!res.ok) throw await res.json()
  return res.json() as T
}

export const backendClient = {
  listAgentTypes: () => call<any[]>('GET', '/internal/agent-types'),
  listAgents: () => call<any[]>('GET', '/internal/agents'),
  createAgent: (body: { agent_type_id: string; name: string }) =>
    call<any>('POST', '/internal/agents', body),
  activateAgent: (id: string) => call<any>('POST', `/internal/agents/${id}/activate`),
  deactivateAgent: (id: string) => call<any>('POST', `/internal/agents/${id}/deactivate`),
  addImConfig: (agentId: string, body: { bot_token: string }) =>
    call<any>('POST', `/internal/agents/${agentId}/im-configs`, body),
  deleteImConfig: (id: string) => call<any>('DELETE', `/internal/im-configs/${id}`),
  chat: (agentId: string, body: { session_id: string; message: string }) =>
    call<any>('POST', `/internal/agents/${agentId}/chat`, body),
```

```tsx
'use client'
import { useState, useEffect, useRef } from 'react'

interface Message { role: 'user' | 'assistant'; text: string }

export function ChatPanel({ agentId }: { agentId: string }) {
  const [sessionId] = useState(() => crypto.randomUUID())
  const [messages, setMessages] = useState<Message[]>([])
  const [input, setInput] = useState('')
  const [loading, setLoading] = useState(false)
  const bottomRef = useRef<HTMLDivElement>(null)

  useEffect(() => {
    bottomRef.current?.scrollIntoView({ behavior: 'smooth' })
  }, [messages])

  async function sendMessage() {
    if (!input.trim() || loading) return
    const text = input.trim()
    setInput('')
    setMessages(m => [...m, { role: 'user', text }])
    setLoading(true)

    try {
      const res = await fetch(`/api/agents/${agentId}/chat`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ session_id: sessionId, message: text }),
      })
      const data = await res.json()
      const reply = data.message?.content?.[0]?.text ?? 'No response'
      setMessages(m => [...m, { role: 'assistant', text: reply }])
    } catch {
      setMessages(m => [...m, { role: 'assistant', text: 'Error: request failed' }])
    } finally {
      setLoading(false)
    }
  }

  return (
    <div style={{ border: '1px solid #ddd', borderRadius: 8, padding: 16, marginTop: 24 }}>
      <h3>Web Test Chat</h3>
      <div style={{ height: 300, overflowY: 'auto', padding: 8, background: '#f9f9f9', marginBottom: 8 }}>
        {messages.map((m, i) => (
          <div key={i} style={{ textAlign: m.role === 'user' ? 'right' : 'left', marginBottom: 8 }}>
            <span style={{
              display: 'inline-block', padding: '6px 12px', borderRadius: 12,
              background: m.role === 'user' ? '#0070f3' : '#e5e5e5',
              color: m.role === 'user' ? 'white' : 'black',
            }}>{m.text}</span>
          </div>
        ))}
        {loading && <div style={{ color: '#999' }}>Thinking...</div>}
        <div ref={bottomRef} />
      </div>
      <div style={{ display: 'flex', gap: 8 }}>
        <input
          value={input}
          onChange={e => setInput(e.target.value)}
          onKeyDown={e => e.key === 'Enter' && sendMessage()}
          placeholder="Type a message..."
          style={{ flex: 1, padding: 8 }}
        />
        <button onClick={sendMessage} disabled={loading}>Send</button>
      </div>
    </div>
  )
}
```
